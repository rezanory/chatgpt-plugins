import canonicalWorker from './index';
import { kernelStatus, listKernels, type WorkerEnv } from './kaggle';
import {
  OFFICIAL_LOCKED_TEST_SCRIPT,
  OFFICIAL_LOCKED_TEST_SCRIPT_SHA256,
  W16_NOTEBOOK_SHA256,
  W16_SOURCE_ZIP_SHA256,
} from './v622-official-locked-test-payload';

type Env = WorkerEnv & { CGP_PROJECT_CONTROL_TOKEN?: string };
type Rec = Record<string, unknown>;
const API_ROOT = 'https://api.kaggle.com/v1/kernels.KernelsApiService';
const OWNER = 'azadka';
const TARGET_SLUG = 'pneumonia-v6-2-2-official-locked-test';
const TARGET_REF = `${OWNER}/${TARGET_SLUG}`;
const TARGET_TITLE = 'PNEUMONIA V6.2.2 OFFICIAL LOCKED TEST';
const ARTIFACT_DATASET = 'trickermark/pneumonia-v6-2-2-finalization-artifacts';
const RAW_DATASET = 'paultimothymooney/chest-xray-pneumonia';
const FINALIZATION_REF = 'azadka/pneumonia-v6-2-2-finalization-master';
const RECONCILIATION_REF = 'azadka/pneumonia-v6-2-2-final-freeze-reconciliation';
const FINAL_MANIFEST_SHA = '2313023819c9eec669d835b99dcc2524b03bb0da236c2ec8f98c4e1221948414';
const RECONCILIATION_SHA = '99a319c8d397342aaefd3c15e0eb8f968a973738f55ef97925a2a16da8c485d8';
const ACTIVE = new Set(['RUNNING','QUEUED','STARTING','PENDING']);
const TERMINAL_BAD = new Set(['ERROR','FAILED','CANCELLED','CANCELED','DEAD']);

function rec(v: unknown): Rec { return v && typeof v === 'object' && !Array.isArray(v) ? v as Rec : {}; }
function json(v: unknown, status=200): Response { return new Response(JSON.stringify(v), {status,headers:{'content-type':'application/json; charset=utf-8','cache-control':'no-store'}}); }
function authorized(r:Request,e:Env):boolean { const t=e.CGP_PROJECT_CONTROL_TOKEN?.trim(); return Boolean(t&&t.length>=32&&r.headers.get('authorization')===`Bearer ${t}`); }
function authHeader(t:string):string { return t.startsWith('KGAT_')?`Bearer ${t}`:`Basic ${btoa(`${OWNER}:${t}`)}`; }
function norm(raw:Rec):string { const s=rec(raw.session); for(const v of [raw.status,raw.statusName,raw.state,raw.sessionStatus,s.status,s.statusName,s.state]) if(typeof v==='string'&&v.trim()) return v.trim().toUpperCase(); return 'UNKNOWN'; }
function allowedOutputUrl(raw:string):URL { const u=new URL(raw),h=u.hostname.toLowerCase(); if(u.protocol!=='https:'||!(h==='api.kaggle.com'||h==='www.kaggle.com'||h==='storage.googleapis.com'||h.endsWith('.kaggleusercontent.com')||h.endsWith('.googleusercontent.com'))) throw new Error('output URL host not allowlisted'); return u; }

async function saveKernel(env:Env):Promise<Rec>{
  const token=env.CGP_KAGGLE_MASTER_TOKEN?.trim(); if(!token) throw new Error('master Kaggle token missing');
  const body:Rec={slug:TARGET_REF,newTitle:TARGET_TITLE,text:OFFICIAL_LOCKED_TEST_SCRIPT,language:'python',kernelType:'script',kernelExecutionType:'SAVE_AND_RUN_ALL',isPrivate:true,enableGpu:true,enableTpu:false,enableInternet:false,kernelDataSources:[FINALIZATION_REF,RECONCILIATION_REF],datasetDataSources:[ARTIFACT_DATASET,RAW_DATASET],competitionDataSources:[],modelDataSources:[]};
  const response=await fetch(`${API_ROOT}/SaveKernel`,{method:'POST',headers:{Authorization:authHeader(token),'Content-Type':'application/json','User-Agent':'chatgpt-v622-official-locked-test/1.0'},body:JSON.stringify(body)});
  const text=await response.text(); let value:Rec={}; try{value=rec(text?JSON.parse(text):{});}catch{throw new Error(`SaveKernel non-JSON HTTP ${response.status}`);}
  const code=typeof value.code==='number'?value.code:undefined; if(!response.ok||(code!==undefined&&code>=400)) throw new Error(String(value.message??`SaveKernel HTTP ${response.status}`).slice(0,1200));
  const badDatasets=Array.isArray(value.invalidDatasetSources)?value.invalidDatasetSources:[]; const badKernels=Array.isArray(value.invalidKernelSources)?value.invalidKernelSources:[];
  if(badDatasets.length||badKernels.length) throw new Error(`SaveKernel rejected data sources: datasets=${JSON.stringify(badDatasets)} kernels=${JSON.stringify(badKernels)}`);
  return value;
}

async function outputs(env:Env):Promise<Rec[]>{
  const token=env.CGP_KAGGLE_MASTER_TOKEN?.trim(); if(!token) throw new Error('master Kaggle token missing');
  const response=await fetch(`${API_ROOT}/ListKernelSessionOutput`,{method:'POST',headers:{Authorization:authHeader(token),'Content-Type':'application/json','User-Agent':'chatgpt-v622-official-locked-test/1.0'},body:JSON.stringify({userName:OWNER,kernelSlug:TARGET_SLUG,pageSize:100})});
  const payload=rec(await response.json()); if(!response.ok) throw new Error(`ListKernelSessionOutput HTTP ${response.status}`); return Array.isArray(payload.files)?payload.files.map(rec):[];
}

async function receipt(env:Env):Promise<Rec|null>{
  const files=await outputs(env);
  const required=['OFFICIAL_LOCKED_TEST_RECEIPT.json','OFFICIAL_LOCKED_TEST_METRICS.json','OFFICIAL_LOCKED_TEST_PREDICTIONS.csv','FINAL_LOCKBOX_CONSUMPTION_RECORD.json','FINAL_LOCKBOX_EVALUATION_MANIFEST.json','FINAL_LOCKBOX_COMPLETION_RECORD.json','MODEL_SIGNATURES.json'];
  for(const name of required){ if(!files.some(x=>String(x.fileName??'').endsWith(name))) return null; }
  const row=files.find(x=>String(x.fileName??'').endsWith('OFFICIAL_LOCKED_TEST_RECEIPT.json')&&typeof x.url==='string'); if(!row) return null;
  const r=await fetch(allowedOutputUrl(String(row.url)),{redirect:'follow'}); if(!r.ok) throw new Error(`receipt download HTTP ${r.status}`); const value=rec(JSON.parse(await r.text())); const metrics=rec(value.metrics);
  if(value.project!=='PNEUMONIA V6.2.2'||value.stage!=='OFFICIAL_LOCKED_TEST'||value.status!=='PASS'||value.n_samples!==624||value.reselection_after_results_allowed!==false||value.next_permitted_stage!=='EXTERNAL_VALIDATION') throw new Error('official locked-test receipt mismatch');
  if(metrics.final_freeze_manifest_sha256!==FINAL_MANIFEST_SHA||metrics.final_freeze_reconciliation_sha256!==RECONCILIATION_SHA||metrics.external_data_used!==false||metrics.n_samples!==624) throw new Error('official locked-test freeze/evidence binding mismatch');
  return {status:'PASS',stage:'OFFICIAL_LOCKED_TEST',n_samples:624,metrics,authorization_record_sha256:value.authorization_record_sha256,evaluation_manifest_sha256:value.evaluation_manifest_sha256,completion_sha256:value.completion_sha256,reselection_after_results_allowed:false,next_permitted_stage:'EXTERNAL_VALIDATION'};
}

async function launch(env:Env):Promise<Rec>{
  const existing=await listKernels(env,'master',TARGET_SLUG,100); const exact=existing.some(x=>String(rec(x).ref??'').toLowerCase()===TARGET_REF.toLowerCase());
  if(exact){
    const status=norm(await kernelStatus(env,'master',TARGET_REF));
    if(ACTIVE.has(status)) return {action:'existing_active_one_shot',target_kernel_ref:TARGET_REF,status,submitted:false};
    if(status==='COMPLETE'){const rc=await receipt(env); if(rc) return {action:'existing_complete_one_shot',target_kernel_ref:TARGET_REF,status,submitted:false,receipt:rc}; throw new Error('locked-test kernel COMPLETE without validated receipt; automatic rerun forbidden');}
    if(TERMINAL_BAD.has(status)) throw new Error(`locked-test kernel already terminal ${status}; automatic rerun forbidden`);
    throw new Error(`locked-test kernel already exists with ambiguous status ${status}; automatic rerun forbidden`);
  }
  const result=await saveKernel(env);
  return {action:'created_and_submitted_once',target_kernel_ref:TARGET_REF,result,submitted:true,kaggle_compute_kind:'OFFICIAL_LOCKED_TEST',gpu:true,internet:false,expected_samples:624,training:false,hpo:false,confirmation:false,reselection:false,external_validation:false,script_sha256:OFFICIAL_LOCKED_TEST_SCRIPT_SHA256,w16_notebook_sha256:W16_NOTEBOOK_SHA256,w16_source_zip_sha256:W16_SOURCE_ZIP_SHA256};
}

export default { async fetch(request:Request,env:Env,ctx:ExecutionContext):Promise<Response>{
  const url=new URL(request.url); if(url.pathname==='/healthz'&&request.method==='GET') return json({service:'v622-official-locked-test',status:'ready',protected:true,target:TARGET_REF,expected_samples:624});
  if(url.pathname.startsWith('/control/v6-2-2/official-locked-test/')&&!authorized(request,env)) return new Response('Forbidden',{status:403});
  try{
    if(url.pathname==='/control/v6-2-2/official-locked-test/launch'&&request.method==='POST') return json({project:'PNEUMONIA V6.2.2',...(await launch(env))});
    if(url.pathname==='/control/v6-2-2/official-locked-test/status'&&request.method==='POST'){const status=norm(await kernelStatus(env,'master',TARGET_REF));const rc=status==='COMPLETE'?await receipt(env):null;return json({project:'PNEUMONIA V6.2.2',target_kernel_ref:TARGET_REF,status,receipt:rc,automatic_rerun_forbidden:true});}
    return canonicalWorker.fetch(request,env,ctx);
  }catch(error){return json({project:'PNEUMONIA V6.2.2',ok:false,error:error instanceof Error?error.message.slice(0,1600):'unknown error',target_kernel_ref:TARGET_REF,automatic_rerun_forbidden:true,training:false,hpo:false,confirmation:false,reselection:false,external_validation:false},502);}
}} satisfies ExportedHandler<Env>;
